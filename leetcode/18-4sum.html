18. 四数之和

题目

给你一个由 n 个整数组成的数组 nums，和一个目标值 target。请你找出并返回满足下述全部条件且不重复的四元组 [nums[a], nums[b], nums[c], nums[d]]（若两个四元组元素一一对应，则认为这两个四元组重复）：

  • 0 <= a, b, c, d < n
  • a、b、c 和 d 互不相同
  • nums[a] + nums[b] + nums[c] + nums[d] == target

可以按 任意顺序 返回答案。

示例 1：

输入：nums = [1,0,-1,0,-2,2], target = 0
输出：[[-2,-1,1,2],[-2,0,0,0],[-1,0,0,1]]
解释：
nums[0] + nums[1] + nums[2] + nums[3] = (-2) + (-1) + 1 + 0 = -2(不等于 0)
nums[0] + nums[1] + nums[3] + nums[4] = (-2) + (-1) + 0 + 1 = -2(不等于 0)
nums[1] + nums[2] + nums[3] + nums[4] = (-1) + 1 + 0 + 1 = 1(不等于 0)
nums[1] + nums[2] + nums[3] + nums[5] = (-1) + 1 + 0 + 2 = 2(不等于 0)
nums[1] + nums[2] + nums[4] + nums[5] = (-1) + 1 + (-2) + 2 = 0
nums[2] + nums[3] + nums[4] + nums[5] = 1 + 0 + (-2) + 2 = 1(不等于 0)
不同的四元组是 [-2,-1,1,2] 和 [-2,0,0,0] 以及 [-1,0,0,1]。

示例 2：

输入：nums = [2,2,2,2,2], target = 8
输出：[[2,2,2,2]]
解释：唯一可能满足题意的四元组之和为 8。

提示：

  • 1 <= nums.length <= 200
  • -10⁹ <= nums[i] <= 10⁹
  • -10⁹ <= target <= 10⁹

方法 1：排序 + 双指针

思路： 本题是「三数之和」的扩展，思路一致：排序后固定前两个数，再用双指针在剩余区间寻找另外两个数，使得四数之和等于 target。

关键点：

  1. 先对数组排序，方便跳过重复元素和进行双指针搜索
  2. 两层循环固定前两个数 nums[i] 和 nums[j]，在 [j+1, n-1] 区间内用双指针 left 和 right 寻找另外两个数
  3. 当四数之和等于 target 时，记录结果并同时跳过左右指针的重复元素
  4. 根据四数之和与 target 的大小关系，移动左指针或右指针
  5. 外层的 i 和 j 都需要跳过重复元素，避免重复的四元组
  6. 剪枝优化： 如果 nums[i] 加上最小的三个数仍大于 target，可以提前结束；如果 nums[i] 加上最大的三个数仍小于 target，可以直接进入下一轮循环
var fourSum = function(nums, target) {
  const res = [];
  const n = nums.length;
  if (n < 4) return res;

  // 数组排序
  nums.sort((a, b) => a - b);

  for (let i = 0; i < n - 3; i++) {
    // 剪枝：nums[i] 加最小的三个数已经大于 target，后面只会更大
    if (nums[i] + nums[i + 1] + nums[i + 2] + nums[i + 3] > target) break;
    // 剪枝：nums[i] 加最大的三个数还小于 target，需要增大 nums[i]
    if (nums[i] + nums[n - 3] + nums[n - 2] + nums[n - 1] < target) continue;
    // 跳过重复的 nums[i]
    if (i > 0 && nums[i] === nums[i - 1]) continue;

    for (let j = i + 1; j < n - 2; j++) {
      // 剪枝：同理对第二层进行剪枝
      if (nums[i] + nums[j] + nums[j + 1] + nums[j + 2] > target) break;
      if (nums[i] + nums[j] + nums[n - 2] + nums[n - 1] < target) continue;
      // 跳过重复的 nums[j]
      if (j > i + 1 && nums[j] === nums[j - 1]) continue;

      let left = j + 1, right = n - 1;
      while (left < right) {
        const sum = nums[i] + nums[j] + nums[left] + nums[right];
        if (sum === target) {
          res.push([nums[i], nums[j], nums[left], nums[right]]);
          // 跳过左右指针的重复元素
          while (left < right && nums[left] === nums[left + 1]) left++;
          while (left < right && nums[right] === nums[right - 1]) right--;
          left++;
          right--;
        } else if (sum < target) {
          left++;
        } else {
          right--;
        }
      }
    }
  }
  return res;
};

时间复杂度：O(n³)，其中 n 是数组的长度。排序的时间复杂度是 O(n log n)，枚举前两个数是 O(n²)，双指针搜索是 O(n)。
空间复杂度：O(log n)，排序使用的额外空间。

方法 2：通用化 k 数之和（递归 + 排序 + 双指针）

思路： 把「四数之和」抽象为「k 数之和」，通过递归减少问题规模。每次固定一个数，将问题转化为「k-1 数之和」，直到 k == 2 时退化为双指针问题。

关键点：

  1. 递归参数包括当前层数 k、起始下标 start、当前目标和 target
  2. 终止条件：k == 2 时使用双指针在 [start, n-1] 区间搜索两数之和等于 target
  3. 当 k > 2 时，遍历 [start, n-k] 区间，固定当前数 nums[i]，递归求解 k-1 数之和为 target - nums[i]
  4. 每层递归都需要跳过重复元素，并对当前固定值进行剪枝
  5. 当 k == n 时，已经固定了所有数字，直接判断总和与目标是否相等
var fourSum = function(nums, target) {
  const res = [];
  nums.sort((a, b) => a - b);

  // kSum 求解从 start 开始、和为 target 的 k 元组
  const kSum = (k, start, target) => {
    if (k === 2) {
      // 双指针求解两数之和
      let left = start, right = nums.length - 1;
      while (left < right) {
        const sum = nums[left] + nums[right];
        if (sum === target) {
          res.push([nums[left], nums[right]]);
          while (left < right && nums[left] === nums[left + 1]) left++;
          while (left < right && nums[right] === nums[right - 1]) right--;
          left++;
          right--;
        } else if (sum < target) {
          left++;
        } else {
          right--;
        }
      }
      return;
    }

    for (let i = start; i <= nums.length - k; i++) {
      // 跳过重复元素
      if (i > start && nums[i] === nums[i - 1]) continue;
      // 递归求解 k-1 元组
      kSum(k - 1, i + 1, target - nums[i]);
    }
  };

  kSum(4, 0, target);
  return res;
};

时间复杂度：O(n³)，与方法 1 相同。递归过程中总共需要处理 n 个数，每层递归将问题规模缩小一层。
空间复杂度：O(n)，递归调用栈的深度为 k（此处为 4），加上排序的额外空间。
